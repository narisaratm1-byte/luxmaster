/** * LUX MASTER - Cloud Sync Backend (Adapted for Numeric Records) * Spreadsheet ID: 1ajTfnOne8JFVCEarIkWmDsW9Q3NGvb-aEUwmBA9oEyA */ const SPREADSHEET_ID = '1ajTfnOne8JFVCEarIkWmDsW9Q3NGvb-aEUwmBA9oEyA'; const SHEET_NAME = "Sheet1"; function doGet() { return HtmlService.createHtmlOutputFromFile('index') .setTitle('LUX MASTER PRO') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL) .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } /** * ฟังก์ชันหลักในการบันทึกข้อมูล * บันทึกทั้งข้อมูลพื้นฐานและตัวเลขที่ใช้ในการคำนวณตามลำดับ m, n, r, p, t, q, L, W */ function processForm(data) { const lock = LockService.getScriptLock(); try { // รอคิว 10 วินาทีเพื่อป้องกันการเขียนทับกัน lock.waitLock(10000); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.getSheets()[0]; // ถ้าไม่เจอชื่อ Sheet1 ให้ใช้แผ่นงานแรก } const timestamp = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss"); const status = (data.status || 'FAIL').toUpperCase(); /** * การเรียงลำดับ rowData (A-V) * บันทึกค่าที่คำนวณและตัวแปรแยกตามคอลัมน์ที่กำหนดไว้ (เริ่ม M=m, N=n...) */ const rowData = [ timestamp, // A: วันที่/เวลา data.location || "", // B: สถานที่ data.floor || "", // C: ชั้น data.room || "", // D: ห้อง Number(data.standard) || 0, // E: มาตรฐาน Lux data.type_label || "", // F: ลักษณะงาน data.bulb || "", // G: สถานะหลอด data.formula_id || "", // H: รหัสสูตร (เช่น F1, F2...) Number(data.lux) || 0, // I: Avg Lux (ผลลัพธ์ที่คำนวณได้) Number(data.max) || 0, // J: Max Number(data.min) || 0, // K: Min status, // L: สถานะ (PASS/FAIL) // --- ส่วนข้อมูลตัวแปรที่ใช้กดคำนวณ (เริ่ม Col M ตามลำดับที่คุณต้องการ) --- data.map_M || "", // M: m data.map_N || "", // N: n data.map_R || "", // O: r data.map_P || "", // P: p (เลขจุดวัดต่างๆ ที่กดคำนวณ) data.map_T || "", // Q: t data.map_Q || "", // R: q data.map_L || "", // S: L data.map_W || "", // T: W data.formula_name || "สูตร 7", // U: ชื่อสูตรหรือเลขที่แสดงการคำนวณ data.image || "" // V: Image Base64 ]; // บันทึกแถวข้อมูลลงในชีท sheet.appendRow(rowData); // ตรวจสอบการระบายสีสถานะ (คอลัมน์ L คือลำดับที่ 12) const lastRow = sheet.getLastRow(); const statusCell = sheet.getRange(lastRow, 12); if (status === "PASS") { statusCell.setFontColor("#10B981").setFontWeight("bold"); } else { statusCell.setFontColor("#EF4444").setFontWeight("bold"); } return { success: true, message: "บันทึกข้อมูลและตัวเลขคำนวณลงชีทสำเร็จ (เริ่มจาก M=m, N=n)" }; } catch (error) { console.error('Error:', error.toString()); return { success: false, message: "เกิดข้อผิดพลาด: " + error.toString() }; } finally { lock.releaseLock(); } } /** * ฟังก์ชันดึงประวัติการบันทึกมาแสดงผล */ function getRecords() { try { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheets()[0]; const lastRow = sheet.getLastRow(); if (lastRow <= 1) return []; const values = sheet.getDataRange().getValues(); return values.slice(1).map((row) => ({ time: row[0] ? row[0].toString() : "", location: row[1] || "", room: row[3] || "", lux: row[8] || 0, status: row[11] || "", image: row[21] || "" })).reverse(); } catch (e) { return []; } }
